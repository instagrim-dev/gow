// DISPOSABLE REVIEW ARTIFACT v3 — assessment-admission-decision recipe, run 3.
// Re-execution of the full C1–C8 scenario against the REMEDIATED tree
// (base 9b901f7: v45 per-evaluation markers, F-2 population-compatible
// authority, v46 attempt→output bindings). Differences from run 2:
//   - C2/C3 use the SAME proposal — the exact layout whose shadowing was
//     finding F-1; the diagnostic distinct-proposal restriction is REMOVED;
//   - C3's tuple is produced by the REGISTERED executed attempt
//     (--procedure equal-denominator), so admission carries the checkable
//     attempt→output link verified by recomputation (v46), replacing the
//     declared bounded-probe note of run 2;
//   - C7's positive branch (preservation of valid current authority after a
//     historical replay) is a REQUIRED assertion, not a recorded outcome:
//     the F-2 fix is in the tree, so displacement is now a failure;
//   - C8's assessment outcome remains DERIVED from observed results.
// Executed once, retained in the evidence bundle, deleted from the tree.

// v3LocalSignature is pcSignature with LOCAL posture: it PRESERVES the mined
// claim `locality == local`, so its admission must not weaken the candidate.
func v3LocalSignature(preservesID string) canon.MechanismSignature {
	sig := pcSignature(preservesID, "", true)
	sig.Posture.Locality = domain.LocalityLocal
	return sig
}

// v3UnknownLocalitySignature is pcSignature with UNKNOWN locality. This is the
// posture the SHARED C2/C3 proposal needs to serve both cases honestly:
//   - at C2 the structural break claim against `locality == local` is
//     UNDECIDABLE (not deterministically refuted), so the evaluation escalates
//     to the model tier and records the single-model-judgment failure the
//     withholding control requires;
//   - after C3's witness admission the materialized member's locality is
//     unknown, which is NOT a decisive counterexample to `locality == local`,
//     so the C6 reassessment over A1 can still end `surviving` (attempt 3.1
//     showed LOCAL posture is deterministically refuted at C2 instead).
func v3UnknownLocalitySignature(preservesID string) canon.MechanismSignature {
	sig := pcSignature(preservesID, "", true)
	sig.Posture.Locality = domain.LocalityUnknown
	return sig
}

// v3TargetIDs parses the persisted generation request payload as the typed
// GenerationRequest and returns the target invariant IDs.
func v3TargetIDs(t *testing.T, ctx context.Context, app *App, dbPath, generationID string) []string {
	t.Helper()
	repo := openTestStore(t, ctx, dbPath)
	defer repo.Close()
	gen, err := repo.GetFrontierGeneration(ctx, generationID)
	if err != nil {
		t.Fatalf("get generation %s: %v", generationID, err)
	}
	invocs, err := repo.ListProviderInvocationsForRun(ctx, gen.RunID)
	if err != nil || len(invocs) == 0 {
		t.Fatalf("list invocations for %s: %v (n=%d)", gen.RunID, err, len(invocs))
	}
	var req provider.GenerationRequest
	if err := json.Unmarshal([]byte(invocs[0].RequestPayload), &req); err != nil {
		t.Fatalf("request payload is not a typed GenerationRequest: %v", err)
	}
	ids := make([]string, 0, len(req.Targets))
	for _, tg := range req.Targets {
		ids = append(ids, tg.InvariantID)
	}
	return ids
}

func v3Targets(t *testing.T, ctx context.Context, app *App, dbPath, generationID, invID string) bool {
	t.Helper()
	for _, id := range v3TargetIDs(t, ctx, app, dbPath, generationID) {
		if id == invID {
			return true
		}
	}
	return false
}

// v3Members returns the sorted signature-ID set of one cluster run.
func v3Members(t *testing.T, ctx context.Context, dbPath, clusterRunID string) []string {
	t.Helper()
	repo := openTestStore(t, ctx, dbPath)
	defer repo.Close()
	rec, err := repo.GetClusterRun(ctx, clusterRunID)
	if err != nil {
		t.Fatalf("get cluster run %s: %v", clusterRunID, err)
	}
	seen := map[string]bool{}
	for _, c := range rec.Clusters {
		for _, m := range c.Members {
			seen[m.SignatureID] = true
		}
	}
	out := make([]string, 0, len(seen))
	for id := range seen {
		out = append(out, id)
	}
	sort.Strings(out)
	return out
}

func TestReviewC1C8V3DisposableScenario(t *testing.T) {
	ctx := context.Background()
	base := time.Date(2026, 9, 12, 15, 0, 0, 0, time.UTC)
	app, dbPath := newRealStoreApp(t, base)
	// Strictly monotonic store clock: predeclaration order is an in-store fact.
	tick := 0
	app.now = func() time.Time { tick++; return base.Add(time.Duration(tick) * time.Second) }
	app.invariantMinerFn = localityMiner{}
	app.challengerFn = counterexampleOnlyChallenger{}

	// ── PREDECLARATION: policy P1r3 + applicability BEFORE any observation ──

		EvidenceCutoff:     base.Format(time.RFC3339), CaseBudget: 8, AttemptBudget: 2, ProviderCallBudget: 0,

			Key: "current-assessment-authority", SemanticRevision: 1,
			Requirement:        "a current decision uses assessments compatible with its declared evidence population and policy; historical replay remains reproducible without restoring obsolete current authority AND without displacing valid current authority",
			AcceptanceCriteria: "C1-C7 including BOTH C7 branches asserted (not recorded) on the real migrated store; C2/C3 on the SAME proposal; C3 attribution via executed attempt binding verified at admission; assessment outcome derived from observed results",
			ApplicabilityRule:  "applies to any candidate with a challenge campaign", PrimaryOwner: "assessment-identity", Mandatory: true,
		}},
	})
	if err != nil {
		t.Fatalf("predeclare policy: %v", err)
	}
	oblID := pol.ObligationIDs["current-assessment-authority@1"]
	if oblID == "" {
		t.Fatalf("obligation id missing from %v", pol.ObligationIDs)
	}
	appl, err := app.DecideReviewApplicability(ctx, ReviewApplicabilityInput{
		DBPath: dbPath, PolicyID: pol.PolicyID, ObligationID: oblID,
		SubjectRef: "the candidate to be mined in this scenario (identity fixed at mining time)",
		Decision:   "applies", Rationale: "the scenario constructs exactly the situation the obligation governs",
		Authorizer: "operator:jmh",
	})
	if err != nil {
		t.Fatalf("predeclare applicability: %v", err)
	}
	t.Logf("PREDECLARED policy=%s obligation=%s applicability=%s (before any observation row)", pol.PolicyID, oblID, appl.ID)

	// ── Setup: discovery population D0, mined candidate ─────────────────────
	problemID, runID, snapshotID := seedProblemAndSnapshot(t, ctx, dbPath, app.now())
	seedAndSign(t, ctx, app, dbPath, problemID, runID, snapshotID, "challenge_population_a.json")

	if err != nil {
		t.Fatalf("cluster build (D0): %v", err)
	}
	if _, err := app.BuildFailureSpace(ctx, FailureSpaceBuildInput{DBPath: dbPath, ProblemID: problemID}); err != nil {
		t.Fatalf("failure space: %v", err)
	}
	mined, err := app.MineInvariants(ctx, InvariantMineInput{DBPath: dbPath, ProblemID: problemID, MinSupport: 2})
	if err != nil || mined.Revision.CandidateCount == 0 {
		t.Fatalf("mine: %v (candidates=%d)", err, mined.Revision.CandidateCount)
	}
	invID := mined.Revision.Candidates[0].ID
	d0Members := v3Members(t, ctx, dbPath, discovery.ClusterRun.ID)
	t.Logf("PIN problem=%s D0=%s |D0|=%d candidate=%s", problemID, discovery.ClusterRun.ID, len(d0Members), invID)

	// ── C1: baseline assess + decision, targets parsed from typed payload ───
	if _, err := app.AuthorInvariantClaim(ctx, AuthorClaimInput{
		DBPath: dbPath, InvariantID: invID, Quantifier: "universal", ClaimRole: "regularity",
		Scope: "failure families of discovery cluster run " + discovery.ClusterRun.ID,
		Note:  "review v3 C1: universal treatment over D0 under P1r3",
	}); err != nil {
		t.Fatalf("C1 author claim: %v", err)
	}
	first, err := app.ChallengeInvariants(ctx, ChallengeInput{DBPath: dbPath, InvariantID: invID})
	if err != nil {
		t.Fatalf("C1 challenge: %v", err)
	}
	r1 := first.Reports[0]
	if r1.StateAfter != "surviving" || r1.AssessmentClusterRunID != discovery.ClusterRun.ID {
		t.Fatalf("C1: want surviving over D0, got %q over %s", r1.StateAfter, r1.AssessmentClusterRunID)
	}
	gen1, err := app.GenerateFrontier(ctx, FrontierGenerateInput{DBPath: dbPath, ProblemID: problemID})
	if err != nil {
		t.Fatalf("C1 generate: %v", err)
	}
	if len(gen1.ExcludedStaleAuthority) != 0 || !v3Targets(t, ctx, app, dbPath, gen1.Generation.ID, invID) {
		t.Fatalf("C1: typed targets must include %s with no stale exclusion; targets=%v", invID, v3TargetIDs(t, ctx, app, dbPath, gen1.Generation.ID))
	}
	t.Logf("C1 PASS campaign=%s targets(typed) include %s", r1.RunID, invID)

	// Mint proposals: [0] LOCAL posture carries C3's bound witness admission
	// (its member decidably conforms to locality==local, so C6 over A1 can
	// survive); [1] UNKNOWN locality is the SHARED C2/C8 proposal — C2's
	// model judgment escalates (undecidable break claim; attempt 3.1 showed
	// LOCAL is deterministically refuted instead) and its later same-proposal
	// witness admission at C8 is the F-1 acceptance case AND the real
	// staleness input (attempt 3.2 showed an unknown-locality member in A1
	// leaves the C6 counterexample search inconclusive — `challenged` — so
	// the same-proposal case moves to A2, where nothing needs survival).
	app.generatorFn = pcGenerator{signatures: []canon.MechanismSignature{
		v3LocalSignature(pcResidueLocality),
		v3UnknownLocalitySignature("domain.number_theory.property.mean_growth_rate"),
	}}
	genMint, err := app.GenerateFrontier(ctx, FrontierGenerateInput{DBPath: dbPath, ProblemID: problemID})
	if err != nil || len(genMint.Generation.Proposals) < 2 {
		t.Fatalf("mint: %v (proposals=%d)", err, len(genMint.Generation.Proposals))
	}
	app.generatorFn = nil
	c3Proposal := genMint.Generation.Proposals[0].ID     // C3 (local, bound attempt)
	sharedProposal := genMint.Generation.Proposals[1].ID // C2 model judgment + C8 same-proposal witness (F-1)

	// ── C2: withheld control (model judgment on the SHARED proposal) ────────
	res, err := app.Evaluate(ctx, EvaluateInput{DBPath: dbPath, ProblemID: problemID, ProposalID: sharedProposal})
	if err != nil {
		t.Fatalf("C2 evaluate: %v", err)
	}
	ev := res.Run.Evaluations[0]
	if ev.Verdict != "failure" || ev.VerificationStrength != "single-model-judgment" {
		t.Fatalf("C2: want model-judged failure, got %q/%q", ev.Verdict, ev.VerificationStrength)
	}
	admitC2, err := app.AdmitEvidence(ctx, AdmitEvidenceInput{DBPath: dbPath, ProblemID: problemID})
	if err != nil || len(admitC2.Admitted) != 0 || len(admitC2.Withheld) != 1 {
		t.Fatalf("C2: rule pass must withhold: %+v err=%v", admitC2, err)
	}
	afterWithhold, err := app.BuildClustering(ctx, ClusterBuildInput{DBPath: dbPath, ProblemID: problemID})
	if err != nil {
		t.Fatalf("C2 recluster: %v", err)
	}
	if got := v3Members(t, ctx, dbPath, afterWithhold.ClusterRun.ID); strings.Join(got, ",") != strings.Join(d0Members, ",") {
		t.Fatalf("C2: withheld observation changed population membership: %v -> %v", d0Members, got)
	}
	t.Logf("C2 PASS withheld=%s on proposal %s; membership set identical to D0", ev.ID, sharedProposal)

	// ── C5: unrelated write must not stale (before the relevant change) ─────
	if _, err := app.PreregisterEpisode(ctx, PreregisterEpisodeInput{
		DBPath: dbPath, ProblemID: problemID, Title: "review v3 C5 unrelated write",
		MapRef: "review-c5-unrelated", Action: "none", Prediction: "none",
		PredictedVerdict: "witness-valid", Note: "outside the declared dependency graph of the assessment",
	}); err != nil {
		t.Fatalf("C5 unrelated write: %v", err)
	}
	gen2, err := app.GenerateFrontier(ctx, FrontierGenerateInput{DBPath: dbPath, ProblemID: problemID})
	if err != nil {
		t.Fatalf("C5 generate: %v", err)
	}
	if len(gen2.ExcludedStaleAuthority) != 0 || !v3Targets(t, ctx, app, dbPath, gen2.Generation.ID, invID) {
		t.Fatalf("C5: unrelated write must not stale the assessment")
	}
	t.Log("C5 PASS unrelated episode write; candidate still targeted")

	// ── C3: EXECUTED attempt, bound and verified at admission (v46) ─────────
	wit, err := app.WitnessCheck(ctx, WitnessCheckInput{
		DBPath: dbPath, ProposalID: c3Proposal,
		Procedure: "equal-denominator", Params: map[string]string{"n": "7"},
	})
	if err != nil {
		t.Fatalf("C3 witness: %v", err)
	}
	if wit.WitnessVerdict != "witness-invalid" {
		t.Fatalf("C3: equal-denominator at n=7 must fail its check, got %q", wit.WitnessVerdict)
	}
	if wit.AttemptBinding == nil || wit.AttemptBinding.TupleCanonical != wit.CanonicalClaim {
		t.Fatalf("C3: the bound tuple must BE the checked claim: %+v vs %q", wit.AttemptBinding, wit.CanonicalClaim)
	}
	admitC3, err := app.AdmitEvidence(ctx, AdmitEvidenceInput{DBPath: dbPath, ProblemID: problemID})
	if err != nil || len(admitC3.Admitted) != 1 {
		t.Fatalf("C3: rule pass must admit the checked failure: %+v err=%v", admitC3, err)
	}
	adm := admitC3.Admitted[0]
	if adm.ProposalID != c3Proposal || adm.EvaluationID != wit.Evaluation.ID {
		t.Fatalf("C3: admission must reference the witness evaluation of the bound proposal: %+v", adm)
	}
	if adm.ObservationKind != ObservationDomainCheckedFailure || adm.AdmittedBy != "rule" || adm.ContentHash == "" {
		t.Fatalf("C3: admitted row must be domain-checked/rule with content hash: %+v", adm)
	}
	if !strings.Contains(adm.Basis, "attempt→output binding verified by recomputation: equal-denominator@") {
		t.Fatalf("C3: admission basis must carry the recomputation-verified link: %q", adm.Basis)
	}
	recluster, err := app.BuildClustering(ctx, ClusterBuildInput{DBPath: dbPath, ProblemID: problemID})
	if err != nil || !recluster.Created {
		t.Fatalf("C3: admitted content must produce population A1: created=%v err=%v", recluster.Created, err)
	}
	a1Members := v3Members(t, ctx, dbPath, recluster.ClusterRun.ID)
	wantA1 := append(append([]string{}, d0Members...), adm.SignatureID)
	sort.Strings(wantA1)
	if strings.Join(a1Members, ",") != strings.Join(wantA1, ",") {
		t.Fatalf("C3: A1 membership must equal D0 + exactly the admitted signature:\n got %v\nwant %v", a1Members, wantA1)
	}
	t.Logf("C3 PASS executed attempt admitted=%s basis carries verified binding; A1 == D0 + admitted signature", adm.SignatureID)

	// ── C4: relevant change — eligibility not inherited from A0 ─────────────
	gen3, err := app.GenerateFrontier(ctx, FrontierGenerateInput{DBPath: dbPath, ProblemID: problemID})
	if err != nil {
		t.Fatalf("C4 generate: %v", err)
	}
	if len(gen3.ExcludedStaleAuthority) != 1 || gen3.ExcludedStaleAuthority[0].InvariantID != invID {
		t.Fatalf("C4: stale assessment must be excluded: %+v", gen3.ExcludedStaleAuthority)
	}
	ex := gen3.ExcludedStaleAuthority[0]
	if ex.AssessedClusterRunID != discovery.ClusterRun.ID || ex.CurrentClusterRunID != recluster.ClusterRun.ID {
		t.Fatalf("C4: exclusion must name A0 and A1: %+v", ex)
	}
	if v3Targets(t, ctx, app, dbPath, gen3.Generation.ID, invID) {
		t.Fatal("C4: stale-authority claim must not be targeted")
	}
	t.Logf("C4 PASS stale exclusion reason=%s", ex.Reason)

	// ── C6: reassessment against A1 ends SURVIVING and drives the decision ──
	second, err := app.ChallengeInvariants(ctx, ChallengeInput{DBPath: dbPath, InvariantID: invID, Population: PopulationLatest})
	if err != nil {
		t.Fatalf("C6 challenge: %v", err)
	}
	r2 := second.Reports[0]
	if r2.AssessmentClusterRunID != recluster.ClusterRun.ID || r2.RunID == r1.RunID {
		t.Fatalf("C6: distinct assessment against A1 required: run=%s assessed=%s", r2.RunID, r2.AssessmentClusterRunID)
	}
	// Precondition for C7's positive branch: REQUIRED, not skipped.
	if r2.StateAfter != "surviving" {
		t.Fatalf("C7 positive-branch setup requires current survival over A1; got %q", r2.StateAfter)
	}
	repo := openTestStore(t, ctx, dbPath)
	defer repo.Close()
	d0row, found, err := repo.GetChallengeAssessmentPopulation(ctx, r1.RunID, invID)
	if err != nil || !found || d0row.AssessmentClusterRunID != discovery.ClusterRun.ID {
		t.Fatalf("C6: historical A0 row must remain reproducible: found=%v %+v err=%v", found, d0row, err)
	}
	gen4, err := app.GenerateFrontier(ctx, FrontierGenerateInput{DBPath: dbPath, ProblemID: problemID})
	if err != nil {
		t.Fatalf("C6 generate: %v", err)
	}
	if len(gen4.ExcludedStaleAuthority) != 0 || !v3Targets(t, ctx, app, dbPath, gen4.Generation.ID, invID) {
		t.Fatalf("C6: surviving-over-current-A1 must restore targeting; excl=%+v", gen4.ExcludedStaleAuthority)
	}
	t.Logf("C6 PASS campaign=%s surviving over A1; candidate targeted again", r2.RunID)

	// ── C7: BOTH branches ASSERTED (F-2 fix in tree) ─────────────────────────
	// Branch 2 (positive, previously the F-2 demonstration): a historical A0
	// replay must NOT displace the valid current authority earned by r2 over
	// A1. With the fix landed this is a required property.
	replay, err := app.ChallengeInvariants(ctx, ChallengeInput{DBPath: dbPath, InvariantID: invID, Population: PopulationDiscovery})
	if err != nil {
		t.Fatalf("C7 replay: %v", err)
	}
	r3 := replay.Reports[0]
	if r3.AssessmentClusterRunID != discovery.ClusterRun.ID {
		t.Fatalf("C7: replay must assess A0, got %s", r3.AssessmentClusterRunID)
	}
	rrow, found, err := repo.GetChallengeAssessmentPopulation(ctx, r3.RunID, invID)
	if err != nil || !found || rrow.PopulationPolicy != PopulationDiscovery {
		t.Fatalf("C7: replay population row must be durable and say discovery: found=%v err=%v", found, err)
	}
	gen5, err := app.GenerateFrontier(ctx, FrontierGenerateInput{DBPath: dbPath, ProblemID: problemID})
	if err != nil {
		t.Fatalf("C7 generate: %v", err)
	}
	c7Preserved := v3Targets(t, ctx, app, dbPath, gen5.Generation.ID, invID)
	c7Detail := fmt.Sprintf("replay=%s state=%q; post-replay targeted=%v staleExcl=%+v",

	if !c7Preserved {
		t.Errorf("C7 POSITIVE BRANCH FAIL: historical replay displaced valid current authority (F-2 regression). %s", c7Detail)
	} else {
		// The operative authority must be the COMPATIBLE campaign (r2 over
		// A1), not the replay: assert the store-level selection directly.
		compat, compatFound, cerr := repo.GetLatestCompatibleAuthority(ctx, invID, recluster.ClusterRun.ID)
		if cerr != nil || !compatFound || compat.RunID != r2.RunID || compat.ToState != "surviving" {
			t.Errorf("C7: operative authority must be the compatible campaign %s (surviving over A1); got %+v found=%v err=%v", r2.RunID, compat, compatFound, cerr)
		} else {
			t.Logf("C7 PASS both branches: obsolete restoration still impossible (r2 authority is compatible+surviving) AND preservation holds. %s; compatible authority=%s", c7Detail, compat.RunID)
		}
	}

	// ── C8: record check + DERIVED assessment; project; real staleness ──────
	chk, err := app.RecordReviewCheck(ctx, ReviewCheckInput{
		DBPath: dbPath, PolicyID: pol.PolicyID, ObligationID: oblID,
		CaseLabel: "C1-C7 combined scenario v3 (both C7 branches asserted; same-proposal C2/C3; bound C3)", ProcedureRef: "docs/reviews/evidence/2026-09-12-c1c8-v3-procedure.go.txt",

		Executor: "go test", Environment: "darwin/local worktree 9b901f7+patch", Mode: "executed", Outcome: "completed",
		OutputRef: "docs/reviews/evidence/2026-09-12-c1c8-v3-run-final.log",
		StartedAt: app.now().Format(time.RFC3339), EndedAt: app.now().Format(time.RFC3339),
		ResourceNote: "single go test invocation; zero provider calls",
	})
	if err != nil {
		t.Fatalf("C8 check: %v", err)
	}
	// Assessment outcome DERIVED from what the run actually observed.
	outcome, argument := "conforms", "C1-C6 and both C7 branches observed conforming on the real migrated remediated store (v45/v46 + F-2 fix); C2/C3 on one shared proposal; C3 admission carries the recomputation-verified attempt→output binding"
	if !c7Preserved {
		outcome = "nonconforms"
		argument = "C7 positive branch demonstrated displacement of valid current authority by a historical replay (F-2 regression): " + c7Detail
	}
	if t.Failed() {
		outcome = "inconclusive"
		argument = "harness recorded assertion failures before assessment; see run log"
	}
	currentRun, foundCur, err := repo.LatestClusterRunForVersions(ctx, problemID, discovery.ClusterRun.SchemaVersion, discovery.ClusterRun.VocabularyVersion)
	if err != nil || !foundCur {
		t.Fatalf("C8: derive current population: %v found=%v", err, foundCur)
	}
	if _, err := app.RecordReviewAssessment(ctx, ReviewAssessInput{
		DBPath: dbPath, PolicyID: pol.PolicyID, ObligationID: oblID,
		ApplicabilityDecisionID: appl.ID, SubjectRef: invID, ContextRef: "population " + currentRun,
		Outcome: outcome, Argument: argument,
		Assessor: "operator:jmh", ProjectRevision: "base 9b901f7 + retained tracked patch + untracked archive",
		EvidenceCutoff: app.now().Format(time.RFC3339),
		Dependencies: []ReviewDependencySpec{{
			Kind: "assessment_population", Ref: currentRun,
			WhyRelevant: "the conformance argument is about authority relative to this current population",
		}},
		CheckAttemptIDs: []string{chk.ID},
	}); err != nil {
		t.Fatalf("C8 assessment: %v", err)
	}
	deps := map[string]string{"assessment_population": currentRun}
	cov1, err := app.GenerateReviewCoverage(ctx, ReviewCoverageInput{DBPath: dbPath, PolicyID: pol.PolicyID, CurrentDependencies: deps, OutPath: "/tmp/review-c1c8-v3-coverage-1.md"})
	if err != nil {
		t.Fatalf("C8 coverage 1: %v", err)
	}
	cov2, err := app.GenerateReviewCoverage(ctx, ReviewCoverageInput{DBPath: dbPath, PolicyID: pol.PolicyID, CurrentDependencies: deps, OutPath: "/tmp/review-c1c8-v3-coverage-2.md"})
	if err != nil {
		t.Fatalf("C8 coverage 2: %v", err)
	}
	if cov1.Document != cov2.Document {
		t.Error("C8: repeated generation from identical records changed the document")
	}
	// The projection must track the RECORDED outcome — this is the decision
	// about the review record, explicitly NOT candidate targetability.
	switch outcome {
	case "conforms":
		if cov1.Decision != "ELIGIBLE_TO_ADVANCE" {
			t.Errorf("C8: conforming mandatory obligation should project eligible, got %s %v", cov1.Decision, cov1.Reasons)
		}
	case "nonconforms":
		if cov1.Decision != "WITHHOLD" {
			t.Errorf("C8: demonstrated nonconformance must project WITHHOLD, got %s %v", cov1.Decision, cov1.Reasons)
		}
	}
	// Real staleness AND the F-1 same-proposal acceptance case: the SHARED
	// proposal already carries C2's withheld model judgment; its later
	// witness-checked failure (executed attempt, n=11) must be independently
	// visible and rule-admitted (v45), preserving the earlier withheld
	// decision. Its admission creates population A2 through the actual
	// pipeline.
	wit2, err := app.WitnessCheck(ctx, WitnessCheckInput{
		DBPath: dbPath, ProposalID: sharedProposal,
		Procedure: "equal-denominator", Params: map[string]string{"n": "11"},
	})
	if err != nil {
		t.Fatalf("C8 witness 2 (same proposal as C2): %v", err)
	}
	admitC8, err := app.AdmitEvidence(ctx, AdmitEvidenceInput{DBPath: dbPath, ProblemID: problemID})
	if err != nil || len(admitC8.Admitted) != 1 {
		t.Fatalf("C8/F-1: the later stronger evaluation of the SAME withheld proposal must be rule-admitted: %+v err=%v", admitC8, err)
	}
	if admitC8.Admitted[0].ProposalID != sharedProposal || admitC8.Admitted[0].EvaluationID != wit2.Evaluation.ID {
		t.Fatalf("C8/F-1: admission must reference the shared proposal's witness evaluation: %+v", admitC8.Admitted[0])
	}
	if admitC8.Admitted[0].EvaluationID == ev.ID {
		t.Fatal("C8/F-1: the admitted row must be the witness evaluation, not the withheld model judgment")
	}
	t.Logf("C8/F-1 PASS same-proposal witness-after-model-judged admitted=%s (withheld=%s preserved)", admitC8.Admitted[0].EvaluationID, ev.ID)
	a2, err := app.BuildClustering(ctx, ClusterBuildInput{DBPath: dbPath, ProblemID: problemID})
	if err != nil || !a2.Created {
		t.Fatalf("C8: A2 build: created=%v err=%v", a2.Created, err)
	}
	currentRun2, foundCur2, err := repo.LatestClusterRunForVersions(ctx, problemID, discovery.ClusterRun.SchemaVersion, discovery.ClusterRun.VocabularyVersion)
	if err != nil || !foundCur2 || currentRun2 == currentRun {
		t.Fatalf("C8: live current population must have advanced: %v %v %s", err, foundCur2, currentRun2)
	}
	cov3, err := app.GenerateReviewCoverage(ctx, ReviewCoverageInput{
		DBPath: dbPath, PolicyID: pol.PolicyID,
		CurrentDependencies: map[string]string{"assessment_population": currentRun2},
		OutPath:             "/tmp/review-c1c8-v3-coverage-3.md",
	})
	if err != nil {
		t.Fatalf("C8 coverage 3: %v", err)
	}
	staleSeen := false
	for _, reason := range cov3.Reasons {
		if reason == "stale_dependency" {
			staleSeen = true
		}
	}
	if cov3.Decision == "ELIGIBLE_TO_ADVANCE" || (outcome == "conforms" && !staleSeen) {
		t.Errorf("C8: pipeline-produced dependency change must stale the assessment: %s %v", cov3.Decision, cov3.Reasons)
	}
	t.Logf("C8 PASS policy=%s recordedOutcome=%s decision=%s deterministic=%v afterRealChange=%s reasons=%v",
		pol.PolicyID, outcome, cov1.Decision, cov1.Document == cov2.Document, cov3.Decision, cov3.Reasons)
}
