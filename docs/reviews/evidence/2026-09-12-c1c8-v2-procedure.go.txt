// DISPOSABLE REVIEW ARTIFACT v2 — assessment-admission-decision recipe, run 2.
// Revision of the 2026-09-12 run-1 procedure after external critique:
//   - policy P1r2 and applicability are persisted BEFORE any observation, on a
//     strictly monotonic store clock (predeclaration is in-store chronology,
//     not fixture-clock assertion);
//   - C1 parses the persisted generation request JSON and inspects the typed
//     `targets` field (no substring matching);
//   - C3 derives the witness tuple from a declared bounded procedure and
//     asserts population MEMBERSHIP SET equality, not counts;
//   - C6 is engineered to end `surviving` over A1 (locality=local admitted
//     signature) and its positive next decision is asserted;
//   - C7 examines BOTH branches: obsolete-restoration prevention AND
//     preservation of valid current authority after a historical replay; the
//     positive branch's precondition is required via t.Fatal, and its outcome
//     is RECORDED (a demonstrated negative is a result, not a harness error);
//   - C8 assessment outcome is gated on the actually observed case results and
//     dependencies are derived from live store state; staleness is produced by
//     a real pipeline change (second admission -> A2), not an injected ref.
// Executed once, retained in the evidence bundle, deleted from the tree.

// v2LocalSignature is pcSignature with LOCAL posture: it PRESERVES the mined
// claim `locality == local`, so its admission must not weaken the candidate.
func v2LocalSignature(preservesID string) canon.MechanismSignature {
	sig := pcSignature(preservesID, "", true)
	sig.Posture.Locality = domain.LocalityLocal
	return sig
}

// v2EqualDenominatorTuple is the declared bounded attempt for C3: the
// equal-denominator probe x=y=z=round(3n/4). Its output IS the checked tuple;
// procedure, parameters, output and checked claim agree by construction.
// For n=7: k=round(21/4)=5, so the attempt emits (5,5,5); 3/5 != 4/7, so the
// probe fails for this instance. The failure refutes THIS bounded attempt.
func v2EqualDenominatorTuple(n int64) (string, int64) {
	k := (3*n + 2) / 4 // round(3n/4) for positive n
	return fmt.Sprintf("%d,%d,%d,%d", n, k, k, k), k
}

// v2TargetIDs parses the persisted generation request payload as the typed
// GenerationRequest and returns the target invariant IDs.
func v2TargetIDs(t *testing.T, ctx context.Context, app *App, dbPath, generationID string) []string {
	t.Helper()
	repo := openTestStore(t, ctx, dbPath)
	defer repo.Close()
	gen, err := repo.GetFrontierGeneration(ctx, generationID)
	if err != nil {
		t.Fatalf("get generation %s: %v", generationID, err)
	}
	invocs, err := repo.ListProviderInvocationsForRun(ctx, gen.RunID)
	if err != nil || len(invocs) == 0 {
		t.Fatalf("list invocations for %s: %v (n=%d)", gen.RunID, err, len(invocs))
	}
	var req provider.GenerationRequest
	if err := json.Unmarshal([]byte(invocs[0].RequestPayload), &req); err != nil {
		t.Fatalf("request payload is not a typed GenerationRequest: %v", err)
	}
	ids := make([]string, 0, len(req.Targets))
	for _, tg := range req.Targets {
		ids = append(ids, tg.InvariantID)
	}
	return ids
}

func v2Targets(t *testing.T, ctx context.Context, app *App, dbPath, generationID, invID string) bool {
	t.Helper()
	for _, id := range v2TargetIDs(t, ctx, app, dbPath, generationID) {
		if id == invID {
			return true
		}
	}
	return false
}

// v2Members returns the sorted signature-ID set of one cluster run.
func v2Members(t *testing.T, ctx context.Context, dbPath, clusterRunID string) []string {
	t.Helper()
	repo := openTestStore(t, ctx, dbPath)
	defer repo.Close()
	rec, err := repo.GetClusterRun(ctx, clusterRunID)
	if err != nil {
		t.Fatalf("get cluster run %s: %v", clusterRunID, err)
	}
	seen := map[string]bool{}
	for _, c := range rec.Clusters {
		for _, m := range c.Members {
			seen[m.SignatureID] = true
		}
	}
	out := make([]string, 0, len(seen))
	for id := range seen {
		out = append(out, id)
	}
	sort.Strings(out)
	return out
}

			t.Errorf("C8: demonstrated nonconformance must project WITHHOLD, got %s %v", cov1.Decision, cov1.Reasons)

	a2, err := app.BuildClustering(ctx, ClusterBuildInput{DBPath: dbPath, ProblemID: problemID})
	if err != nil || !a2.Created {
		t.Fatalf("C8: A2 build: created=%v err=%v", a2.Created, err)
	}
	currentRun2, foundCur2, err := repo.LatestClusterRunForVersions(ctx, problemID, discovery.ClusterRun.SchemaVersion, discovery.ClusterRun.VocabularyVersion)
	if err != nil || !foundCur2 || currentRun2 == currentRun {
		t.Fatalf("C8: live current population must have advanced: %v %v %s", err, foundCur2, currentRun2)
	}
	cov3, err := app.GenerateReviewCoverage(ctx, ReviewCoverageInput{
		DBPath: dbPath, PolicyID: pol.PolicyID,
		CurrentDependencies: map[string]string{"assessment_population": currentRun2},
		OutPath:             "/tmp/review-c1c8-v2-coverage-3.md",
	})
	if err != nil {
		t.Fatalf("C8 coverage 3: %v", err)
	}
	staleSeen := false
	for _, reason := range cov3.Reasons {
		if reason == "stale_dependency" {
			staleSeen = true
		}
	}
	if cov3.Decision == "ELIGIBLE_TO_ADVANCE" || (outcome == "conforms" && !staleSeen) {
		t.Errorf("C8: pipeline-produced dependency change must stale the assessment: %s %v", cov3.Decision, cov3.Reasons)
	}
	t.Logf("C8 PASS policy=%s recordedOutcome=%s decision=%s deterministic=%v afterRealChange=%s reasons=%v",
		pol.PolicyID, outcome, cov1.Decision, cov1.Document == cov2.Document, cov3.Decision, cov3.Reasons)
}
